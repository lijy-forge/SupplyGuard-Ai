1. 实验相关接口
1.1 获取实验信息
接口 : GET /api/experiments/{exp_id}
数据内容 :
interface ApiExperiment {
  id: string;           // 实验ID
  name: string;         // 实验名称
  num_day: number;      // 实验天数
  status: string;       // 实验状态
  cur_day: number;      // 当前天数
  cur_t: number;        // 当前时间（秒）
  config: string;       // 实验配置
  error: string;        // 错误信息
  input_tokens: number; // 输入token数
  output_tokens: number;// 输出token数
  created_at: string;   // 创建时间
  updated_at: string;   // 更新时间
}
1.2 获取实验时间线
接口 : GET /api/experiments/{exp_id}/timeline
数据内容 :
interface ApiTime {
  day: number;  // 天数
  t: number;    // 时间
}
2. Agent相关接口
2.1 获取所有Agent档案
接口 : GET /api/experiments/{exp_id}/agents/-/profile
数据内容 :
interface ApiAgentProfile {
  id: number;      // Agent ID
  name: string;    // Agent名称
  profile: any;    // Agent档案（JSON对象）
}
2.2 获取特定Agent档案
接口 : GET /api/experiments/{exp_id}/agents/{agent_id}/profile
数据内容 : 同上 ApiAgentProfile
2.3 获取所有Agent状态（按时间）
接口 : GET /api/experiments/{exp_id}/agents/-/status?day={day}&t={t}
数据内容 :
interface ApiAgentStatus {
  id: number;           // Agent ID
  day: number;          // 天数
  t: number;            // 时间
  lng?: number;         // 经度（可选）
  lat?: number;         // 纬度（可选）
  parent_id?: number;   // 父Agent ID（可选）
  action: string;       // 行为
  status: any;          // 状态（JSON对象）
  created_at: string;   // 创建时间
}
2.4 获取特定Agent状态历史
接口 : GET /api/experiments/{exp_id}/agents/{agent_id}/status
数据内容 : 同上 ApiAgentStatus 数组

2.5 获取特定Agent对话记录
接口 : GET /api/experiments/{exp_id}/agents/{agent_id}/dialog
数据内容 :
interface ApiAgentDialog {
  id: number;         // Agent ID
  day: number;        // 天数
  t: number;          // 时间
  type: number;       // 对话类型（0=思考，1=交谈，2=用户）
  speaker: string;    // 发言者
  content: string;    // 内容
  created_at: string; // 创建时间
}
3. 全局提示接口
3.1 获取全局提示（Market Insight）
接口 : GET /api/experiments/{exp_id}/prompt?day={day}&t={t}
数据内容 :
interface ApiGlobalPrompt {
  day: number;        // 天数
  t: number;          // 时间（秒）
  prompt: string;     // 提示内容
  created_at: string; // 创建时间
}
